\usepackage[most]{tcolorbox}

% Keep highlighted code blocks visually separated from surrounding prose/lists.
% \addvspace collapses adjacent vertical skips instead of accumulating them.
\renewenvironment{Shaded}
  {\par\addvspace{0.6\baselineskip}\begin{snugshade}}
  {\end{snugshade}\par\addvspace{0.6\baselineskip}}

% Pandoc makes the table entries clickable, but does not bookmark the TOC page itself.
\let\algorithmoldtableofcontents\tableofcontents
\renewcommand{\tableofcontents}{%
  \pdfbookmark[0]{\contentsname}{algorithm-toc}%
  \algorithmoldtableofcontents
}

\tcbset{
  callout/.style={
    enhanced,
    breakable,
    boxrule=0pt,
    leftrule=2.6pt,
    arc=0.8mm,
    outer arc=0.8mm,
    left=3mm,
    right=3mm,
    top=1.5mm,
    bottom=2mm,
    before skip=9pt,
    after skip=9pt,
    fonttitle=\bfseries,
    colbacktitle=white,
    attach boxed title to top left={xshift=2mm,yshift=-2mm},
    boxed title style={
      boxrule=0pt,
      colback=white,
      left=1mm,
      right=1mm,
      top=0pt,
      bottom=0pt
    }
  }
}
